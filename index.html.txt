 return {high:hi,low:lo,up:ih>il};

// NEU: Swing aus mehr Kerzen - aber nur aus den letzten X% für Fibo, sonst ist der Swing zu alt
function findSwingSmart(candles){
 // Für Fibo: nimm den Swing aus den letzten 300 Kerzen (oder allen, wenn weniger da sind)
 // So ist der Swing aktuell, aber trotzdem aus viel mehr Daten als vorher 100
 var look=Math.min(300, candles.length);
 var slice=candles.slice(-look);
 var hi=-1e18,lo=1e18,ih=0,il=0;
 for(var i=0;i<slice.length;i++){var x=slice[i];if(x.high>hi){hi=x.high;ih=i;}if(x.low<lo){lo=x.low;il=i;}}
 return {high:hi,low:lo,up:ih>il,look:look};
}
function zigzag(c,pct){
 var z=[],lh=c[0].high,ll=c[0].low,lhi=0,lli=0,dir=0;
 for(var i=1;i<c.length;i++){var x=c[i];
  if(dir>=0){if(x.high>lh){lh=x.high;lhi=i;}if(x.low<lh*(1-pct)){z.push({i:lhi,p:lh,t:'H',time:c[lhi].time});dir=-1;ll=x.low;lli=i;}}
  else{if(x.low<ll){ll=x.low;lli=i;}if(x.high>ll*(1+pct)){z.push({i:lli,p:ll,t:'T',time:c[lli].time});dir=1;lh=x.high;lhi=i;}}
 }
 z.push(dir>=0?{i:lhi,p:lh,t:'H',time:c[lhi].time}:{i:lli,p:ll,t:'T',time:c[lli].time});return z;
}

  if(sc<=bs)continue;bs=sc;

function calcATR(candles, p){
 p=p||14;
 var trs=[];
 for(var i=1;i<candles.length;i++){
  var h=candles[i].high, l=candles[i].low, pc=candles[i-1].close;
  trs.push(Math.max(h-l, Math.abs(h-pc), Math.abs(l-pc)));
 }
 if(trs.length<p) return 0;
 var atr=0; for(var j=0;j<p;j++) atr+=trs[j]; atr/=p;
 for(var k=p;k<trs.length;k++) atr=(atr*(p-1)+trs[k])/p;
 return atr;
}
// NEU: VWAP aus mehr Kerzen - je nach Timeframe 100-200
function calcVWAP(candles, look){
 look=look||200;
 var n=Math.min(look, candles.length);
 var sumPV=0, sumV=0;
 var series=[];
 for(var i=candles.length-n;i<candles.length;i++){
  var c=candles[i];
  var tp=(c.high+c.low+c.close)/3;
  var v=c.vol||1;
  sumPV+=tp*v; sumV+=v;
  series.push({time:c.time, value: sumV>0? sumPV/sumV : tp});
 }
 return {value: sumV>0? sumPV/sumV: candles[candles.length-1].close, series: series, look:n};
}
// NEU: S/R aus 150-300 Kerzen statt 100 - mit Stärke
function calcSR(candles, look){
 look=look||300;
 var slice=candles.slice(-look);
 var price=candles[candles.length-1].close;
 var levels=[];
 for(var i=0;i<slice.length;i++){
  levels.push(slice[i].high); levels.push(slice[i].low);
 }
 var clusters=[];
 for(var j=0;j<levels.length;j++){
  var lvl=levels[j]; var found=null;
  for(var k=0;k<clusters.length;k++){ if(Math.abs(clusters[k].price-lvl)/lvl<0.008){ found=clusters[k]; break; } }
  if(found){ found.count++; found.price=(found.price*(found.count-1)+lvl)/found.count; }
  else clusters.push({price:lvl, count:1});
 }
 clusters.sort(function(a,b){return b.count-a.count;});
 var supports=[], resistances=[];
 for(var m=0;m<clusters.length;m++){
  var c2=clusters[m];
  if(c2.count<3) continue;
  if(c2.price<price) supports.push(c2);
  else if(c2.price>price) resistances.push(c2);
 }
 supports.sort(function(a,b){return b.price-a.price;});
 resistances.sort(function(a,b){return a.price-b.price;});
 return {support: supports.slice(0,3), resistance: resistances.slice(0,3), nearestSupport: supports[0]||null, nearestResistance: resistances[0]||null, totalLevels: clusters.length, look:look};
}
function detectSweep(candles){
 if(candles.length<20) return 'KEIN';
 var last=candles[candles.length-1];
 var prevHigh=Math.max.apply(null, candles.slice(-20,-1).map(function(x){return x.high;}));
 var prevLow=Math.min.apply(null, candles.slice(-20,-1).map(function(x){return x.low;}));
 if(last.high>prevHigh && last.close<prevHigh) return 'BEARISH SWEEP - Hoch abgeholt, dann runter = Fake-Ausbruch nach oben';
 if(last.low<prevLow && last.close>prevLow) return 'BULLISH SWEEP - Tief abgeholt, dann hoch = Fake-Ausbruch nach unten';
 if(last.high>prevHigh && last.close>prevHigh) return 'ECHTER AUSBRUCH NACH OBEN';
 if(last.low<prevLow && last.close<prevLow) return 'ECHTER AUSBRUCH NACH UNTEN';
 return 'KEIN SWEEP';
}
