function zigzag(c,pct){
 let z=[],lh=c[0].high,ll=c[0].low,lhi=0,lli=0,dir=0;
 for(let i=1;i<c.length;i++){let x=c[i];

  else{if(x.low<ll){ll=x.low;lli=i;}if(x.high>ll*(1+pct)){z.push({i:lli,p:ll,t:'T'});dir=1;lh=x.high;lhi=i;}}
 }
 z.push(dir>=0?{i:lhi,p:lh,t:'H'}:{i:lli,p:ll,t:'T'});return z;
}

 else if(state.elliott&&state.elliott.welle.includes('WELLE 2')){blocked=true;grund='Elliott in Welle 2 - warte.';}
 else if(state.abc&&state.abc.welle.includes('WELLE B')){blocked=true;grund='ABC in Welle B - danach kommt C runter.';}
 else if(sc>=65&&sig==='BULLISCH'){ent='LONG';grund='Score '+sc+' + BULLISCH. Rückenwind!';entry=price;stop=fib['1'];target=fib['0.786'];}
 else if(sc<=35&&sig==='BÄRISCH'){ent='SHORT';grund='Score '+sc+' + BÄRISCH.';entry=price;stop=fib['0'];target=fib['1'];}
 else if(sc>=80){ent='LONG';grund='Score '+sc+' extrem stark.';entry=price;stop=fib['1'];target=fib['0.618'];}
 else if(sc<=20){ent='SHORT';grund='Score '+sc+' extrem schwach.';entry=price;stop=fib['0'];target=fib['1'];}
 else{grund=sc>=45?'Score '+sc+' zu schwach für einen Trade.':sig==='BODENBILDUNG'?'Signal ist BODENBILDUNG - warten.':'Keine klare Kante.';}

async function scanAll(){

 let rows=[];
 for(let c of COINS){
  let r=await getKlines(c+'USDT','1d',86400);
  if(!r.data.length){rows.push('<tr><td><b>'+c+'</b></td><td>-</td><td>Keine Daten</td></tr>');continue;}
  let closes=r.data.map(x=>x.close);let rr=rsiCalc(closes,14);
  let sw=findSwing(r.data);let fib=calcFibo(sw.high,sw.low,sw.up);
  let sig=closes[closes.length-1]>fib['0.382']?'BULLISCH':'BÄRISCH';
  rows.push('<tr><td><b>'+c+'</b></td><td>'+closes[closes.length-1].toFixed(dec(c))+' $</td><td>RSI '+rr.toFixed(1)+' • '+sig+'</td></tr>');
  await new Promise(x=>setTimeout(x,120));
 }
 $('allBox').innerHTML='<table><tr><th>Coin</th><th>Preis</th><th>RSI / Signal</th></tr>'+rows.join('')+'</table>';
}
